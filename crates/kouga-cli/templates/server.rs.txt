#[tokio::main(flavor = "multi_thread")]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let address = std::env::var("KOUGA_BIND").unwrap_or_else(|_| "127.0.0.1:3000".to_owned());
    let listener = tokio::net::TcpListener::bind(&address).await?;
    println!("Kouga HTTP listening on {}", listener.local_addr()?);
    axum::serve(listener, APP_CRATE::router().with_state(())).await?;
    Ok(())
}
